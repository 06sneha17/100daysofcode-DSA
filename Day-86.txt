#include <stdio.h>

int integerSquareRoot(int n) {
    int low = 0, high = n, mid, ans = 0;

    while (low <= high) {
        mid = (low + high) / 2;

        if (mid * mid == n) {
            return mid;  // perfect square
        }
        else if (mid * mid < n) {
            ans = mid;   // store possible answer
            low = mid + 1;
        }
        else {
            high = mid - 1;
        }
    }

    return ans;
}

int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);

    printf("%d\n", integerSquareRoot(n));

    return 0;
}
